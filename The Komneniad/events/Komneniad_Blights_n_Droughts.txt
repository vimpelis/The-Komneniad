    }

country_event = {
    id = 179001
    title = "Potato Blight Spreads"
    desc = "Scary!"
    picture = "ZPD"

    is_triggered_only = yes

    option = {
        name = "Oof"
		any_country = {
			any_owned = {
				limit = { has_province_modifier = potato_blight_special }
				any_neighbor_province = { limit = { continent = europe } add_province_modifier = { name = potato_blight_special duration = 730 } }
				sea_zone = {
					random_neighbor_province = { limit = { continent = europe } add_province_modifier = { name = potato_blight_special duration = 730 } }
				}
			}
		}
		random_owned = {
			limit = { owner = { NOT = { month = 10 } } }
			owner = { country_event = { id = 179001 days = 18 } }
		}
    }
}

province_event = {
    id = 179002
    title = "Potato Blight arrives in $PROVINCENAME$"
    desc = "179002DESC"

    trigger = {
		OR = {
			province_id = 442
			province_id = 423
			province_id = 420
			province_id = 418
			province_id = 402
			province_id = 388
			province_id = 375
			province_id = 303
		}
		owner = { year = 1845 month = 6 NOT = { has_global_flag = 1845_potato_blight } }
    }

    fire_only_once = yes

    mean_time_to_happen = {
        days = 15
    }

    option = {
        name = "Hopefully it won't amount to much."
		owner = { country_event = 179000 }
		add_province_modifier = { name = potato_blight_special duration = 730 }
		set_global_flag = 1845_potato_blight
    }
}
