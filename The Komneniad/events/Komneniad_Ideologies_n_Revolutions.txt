country_event = {
    id = 180000
    title = "The Spring of Nations"
    desc = "180000DESC"
    picture = "spring_of_nations"

    trigger = {
		year = 1848
		OR = {
			tag = FRA
			tag = OCC
			tag = BYZ
			tag = GRE
			tag = BAV
			tag = DAN
			tag = ARE
			tag = ZPD
			tag = LOM
		}
    }

    fire_only_once = yes

    mean_time_to_happen = {
        days = 15
    }
	
	major = yes

    option = {
        name = "Aux armes, citoyens!"
		any_country = {
			limit = { capital_scope = { continent = europe } }
			any_pop = { consciousness = 1 }
		}
		set_global_flag = spring_of_nations_flag
    }
}

country_event = {
    id = 180001

    desc = "180001DESC"
    picture = "spring_of_nations"

    trigger = {
		year = 1849
		is_greater_power = yes
		war = no
		NOT = {
			OR = {
				tag = FRA
				tag = BYZ
				tag = GRE
			}
		}
    }

    fire_only_once = yes

    mean_time_to_happen = {
        days = 30
    }

    option = {
        name = "Let's try this again."
		any_country = {
			limit = { capital_scope = { continent = europe } }
			country_event = { id = 180002 days = 3 }
		}
		set_country_flag = hidden_tooltip
		random_owned = { limit = { owner = { has_country_flag = hidden_tooltip } }
			owner = {
				country_event = { id = 180003 days = 15 }
			}
		}
		clr_country_flag = hidden_tooltip
    }
}
